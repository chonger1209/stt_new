# Caps Lock 检测器配置文件
# 配置说明版本: 2
#
# 使用说明：
# 1. 本文件使用 UTF-8 编码；以 # 开头的整行是说明，不是设置。
# 2. 修改并保存后，点击程序标题栏的 ⟳ 或右键菜单“刷新”生效。
# 3. true 表示开启/显示，false 表示关闭/隐藏。
# 4. 颜色、窗口、计时等全局设置必须放在第一个 [进程名称] 之前。
# 5. 程序退出时只更新 window_x / window_y，不重置其他设置。
# 6. 行内分号 ; 后面作为注释，名称等配置值中不要包含分号。

# 颜色设置
# color_caps_on: Caps Lock 开启时的背景色，默认 #fa6666（红色）；可填 #RRGGBB 颜色值。
color_caps_on = #fa6666
# color_caps_off: Caps Lock 关闭时的背景色，默认 #4CAF50（绿色）。
color_caps_off = #4CAF50
# color_titlebar: 标题栏及标题栏按钮背景色，默认 #2c3e50。
color_titlebar = #2c3e50

# 窗口设置
# window_width: 窗口宽度，单位像素，正整数；默认 300。
window_width = 300
# window_height: 统计面板收起时的窗口高度，单位像素，正整数；默认 240。
window_height = 240
# window_x: 窗口横坐标，单位像素；默认 -1。横/纵坐标任一为 -1 时居中；退出时自动保存位置。
window_x = -1
# window_y: 窗口纵坐标，单位像素；默认 -1。多显示器坐标可以超过主屏幕宽度，也可以为负数。
window_y = -1
# always_on_top: 是否始终置顶：true=置顶，false=普通窗口；默认 true。
always_on_top = true

# 自动大小写切换
# software_list: 自动开启大写锁定的软件关键词，用英文逗号分隔，忽略大小写。
# 匹配应用显示名称或窗口标题；切换到不匹配的软件时自动关闭大写锁定。
# 默认 CAXA,CAD,SOLIDWORKS；留空表示切换到任何软件时都关闭大写锁定，并非停用自动切换。
software_list = CAXA,CAD,SOLIDWORKS

# 时间流与统计刷新
# grid_second_per_block: 时间流每个色块代表的秒数，也是采样/定时刷新间隔；正整数，默认 30。
# 不同应用的短片段共同累计，满设定秒数才生成一格，不足部分继续累计。
# 色块颜色取该格累计结束时最后使用的应用；切换应用不会额外产生格子。
grid_second_per_block = 30
# screen_time_refresh_frequency: 统计面板刷新频率，单位“次/10秒”，正整数，默认 10。
# 10=每秒一次，2=每5秒一次，1=每10秒一次；当前最快每秒一次，超过10不会更快。
# 只影响统计面板刷新，不改变每200毫秒检测前台窗口的频率。
screen_time_refresh_frequency = 10

# 番茄倒计时
# timer_duration_minutes: 番茄倒计时时长，单位分钟，正整数；默认 25。
# 修改后点击“刷新”：尚未开始时立即更新；正在运行或暂停时保留剩余时间，重置后用新时长。
timer_duration_minutes = 25

# 千问休眠设置
# qwen_idle_timeout_minutes: 连续闲置后卸载模型的时间，单位分钟，正整数；默认 120（2小时）。
# 卸载模型释放其内存和显存占用，保留接收服务，本地或局域网请求可自动唤醒。
# 执行或排队任务时不休眠；最后一个任务结束后重新计时，状态查询不延时。
# 提前1分钟提醒，点击重置可重新计算完整时长；修改后点击“刷新”生效。
# 无效值使用120分钟并记录原因；STT退出后不再执行闲置管理。
qwen_idle_timeout_minutes = 120

# 千问连接与启动（路径参数下次启动服务生效）
# qwen_enabled: 是否启用千问管理；true/false。false 不查询、不启动、不自动休眠，普通统计照常。
qwen_enabled = true
# qwen_prepare_service_on_start: STT 启动时是否准备接收服务；true/false。不主动加载模型。
qwen_prepare_service_on_start = true
# qwen_toolkit_path: 完整管理工具包目录，内含 scripts/manager.py 和 lifecycle.py；留空自动查找。示例 E:/AI/qwen-management。
qwen_toolkit_path =
# qwen_base_url: 本机服务地址，例 http://127.0.0.1:8765/v1；留空沿用工具包。不能填写远程 IP。
qwen_base_url =
# qwen_model_id: 服务公布的模型 ID，不一定等于模型文件夹名称；留空沿用工具包。
qwen_model_id =
# qwen_python_path: 模型环境的 python.exe 完整路径；例 E:/AI/Strata/.venv/Scripts/python.exe。留空沿用工具包。
qwen_python_path =
# qwen_working_directory: 模型服务工作目录，例 E:/AI/Strata；留空沿用工具包。
qwen_working_directory =
# qwen_launch_args: 启动参数，填 JSON 字符串数组，建议路径用 /；例 ["serve/server.py", "--port", "8765"]。
# 必须复制已经验证可用的参数；模型路径通常也在其中。留空沿用，[] 表示明确清空参数。
qwen_launch_args =
# qwen_startup_timeout_seconds: 整个启动操作的总等待预算，单位秒，正整数；默认900。
qwen_startup_timeout_seconds = 900

# 进程配置（可选，不是白名单）
# 没有配置的程序也会正常记录，默认在统计和时间流中都显示；通常显示进程文件名。
# 只有需要改显示名称或隐藏某个程序时，才需要添加/修改对应条目。
# 格式：[进程名称.exe]，例如 [chrome.exe]；进程名匹配忽略大小写。
# display_name: 在界面中显示的名称；省略时使用本节的进程名称。
# show_in_screen_time: 是否在当天使用时长统计中显示，true/false；省略默认 true。
# show_in_time_stream: 是否在彩色时间流中显示，true/false；省略默认 true。
# false 只隐藏对应视图，不停止记录，也不会删除已有历史数据。
# 不同程序建议使用不同的 display_name；相同名称会合并到同一个统计项目。
# LockApp.exe 是锁屏程序：首次生成配置时，默认在统计中隐藏，在时间流中保留。
